# Go 语法高亮示例

以下 `go.go` 示例使用 ` ```go ` 围栏代码块渲染：

```go
// Go 语言：编译型、并发友好的静态语言，以简洁、工程化和 goroutine 闻名。
package main

// ===== import 分组：标准库统一放在一个分组中 =====
import (
	"errors"
	"fmt"
	"sync"
)

// ===== 变量与类型 =====
var globalCount int = 100 // 包级变量，显式类型

// ===== 常量与 iota =====
const (
	Sunday    = iota // 0
	Monday           // 1
	Tuesday          // 2
)

// ===== 结构体与方法 =====
type Point struct {
	X, Y float64
}

// 值接收者方法
func (p Point) Distance() float64 {
	return p.X*p.X + p.Y*p.Y
}

// 指针接收者方法：可修改接收者
func (p *Point) Scale(factor float64) {
	p.X *= factor
	p.Y *= factor
}

// ===== 接口 =====
type Shape interface {
	Area() float64
}

type Circle struct {
	Radius float64
}

func (c Circle) Area() float64 {
	return 3.14159 * c.Radius * c.Radius
}

// ===== 函数：多返回值、命名返回 =====
func divide(a, b float64) (q, r float64, err error) {
	if b == 0 {
		err = errors.New("divide by zero")
		return // 裸 return，使用命名返回值
	}
	q = a / b
	r = a - b*float64(int(a/b))
	return
}

// ===== 泛型：类型参数（Go 1.18+） =====
func Max[T int | float64](a, b T) T {
	if a > b {
		return a
	}
	return b
}

func main() {
	// 短声明 :=
	name := "Go"
	version := 1.22
	fmt.Printf("%s %.2f\n", name, version)

	// 指针与解引用
	x := 10
	p := &x
	*p = 20
	fmt.Println("x =", x) // 20

	// 控制流：if（可带初始化语句）
	if n := 15; n%2 == 0 {
		fmt.Println("even")
	} else {
		fmt.Println("odd")
	}

	// 控制流：for（Go 只有 for 一种循环）
	for i := 0; i < 3; i++ {
		fmt.Println("i =", i)
	}

	// 控制流：switch（默认不穿透，无需 break）
	switch day := Monday; day {
	case Sunday:
		fmt.Println("Sunday")
	case Monday, Tuesday:
		fmt.Println("start of week")
	default:
		fmt.Println("other")
	}

	// 切片与映射
	nums := []int{1, 2, 3}
	nums = append(nums, 4)
	scores := map[string]int{"alice": 90, "bob": 85}
	scores["carol"] = 70
	fmt.Println("nums:", nums, "scores:", scores)

	// 错误处理：多返回值 + if err
	q, _, err := divide(10, 0)
	if err != nil {
		fmt.Println("error:", err)
	} else {
		fmt.Println("q =", q)
	}

	// defer：LIFO 延迟执行（常用于资源释放）
	defer fmt.Println("deferred last (runs at return)")

	// 接口多态
	var s Shape = Circle{Radius: 2}
	fmt.Printf("area = %.2f\n", s.Area())

	// 泛型调用
	fmt.Println("Max int:", Max(3, 7), "Max float:", Max(2.5, 1.5))

	// goroutine 与 channel：简单同步
	ch := make(chan int)
	go func() { ch <- 42 }()
	fmt.Println("from channel:", <-ch)

	// 带缓冲 channel + WaitGroup：多 worker 模型
	jobs := make(chan int, 3)
	var wg sync.WaitGroup
	for w := 0; w < 2; w++ {
		wg.Add(1)
		go func(id int) {
			defer wg.Done()
			for j := range jobs {
				fmt.Printf("worker %d got job %d\n", id, j)
			}
		}(w)
	}
	for j := 1; j <= 3; j++ {
		jobs <- j
	}
	close(jobs) // 关闭后 range 退出
	wg.Wait()
}
```
