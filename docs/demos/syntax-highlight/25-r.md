# R 语法高亮示例

以下 `r.R` 示例使用 ` ```r ` 围栏代码块渲染：

```r
# R — 用于统计计算与数据可视化的语言，向量为核心数据单位。

# ============================================================
# 注释、赋值与基本类型
# ============================================================
# 注释以 # 开头，没有多行注释语法
x <- 10            # 推荐的赋值运算符 <-
y = 20             # = 也能赋值，惯例在顶层用 <-
name <- "R"        # 字符型
flag <- TRUE       # 逻辑型（TRUE/FALSE，可用缩写 T/F）
miss <- NA         # NA 表示缺失值
nothing <- NULL    # NULL 表示“不存在”
big <- 1e6         # 数值（双精度）

# ============================================================
# 向量：c() 与向量化运算
# ============================================================
v <- c(1, 2, 3, 4, 5)             # 数值向量
chars <- c("a", "b", "c")         # 字符向量
logs <- c(TRUE, FALSE, TRUE)      # 逻辑向量
seq_v <- 1:10                     # 整数序列
seq_p <- seq(0, 1, by = 0.25)     # 等差数列
rep_v <- rep("x", times = 3)      # 重复

# 向量化运算（逐元素）
squared <- v^2
plus_ten <- v + 10
cond <- v > 2                     # 返回逻辑向量
v[cond]                           # 用逻辑向量筛选

# ============================================================
# 矩阵与数组
# ============================================================
m <- matrix(1:12, nrow = 3, ncol = 4)   # 按列填充
dim(m)                                  # 维度 (3, 4)
t(m)                                    # 转置
rowSums(m); colMeans(m)
m %*% t(m)                              # 矩阵乘法

arr <- array(1:24, dim = c(2, 3, 4))    # 三维数组

# ============================================================
# 列表：异构数据容器（类似其他语言的 dict/对象）
# ============================================================
lst <- list(
  id = 1L,
  name = "Alice",
  scores = c(90, 85, 88),
  nested = list(a = 1, b = 2)
)

# ============================================================
# 数据框：表格数据（每列可不同类型）
# ============================================================
df <- data.frame(
  id = 1:3,
  name = c("Alice", "Bob", "Carol"),
  score = c(90.5, 85.0, 88.5),
  stringsAsFactors = FALSE
)
nrow(df); ncol(df); names(df)

# ============================================================
# 因子：分类变量（名义 / 有序）
# ============================================================
size <- factor(c("S", "M", "L", "M", "S"),
               levels = c("S", "M", "L"))     # 名义因子
grade <- factor(c("A", "B", "A", "C"),
                levels = c("C", "B", "A"),
                ordered = TRUE)               # 有序因子
table(size)                                    # 频数统计

# ============================================================
# 索引：[] / [[]] / $
# ============================================================
v[c(1, 3)]            # 向量按位置取值
v[v > 2]              # 逻辑筛选
m[2, 3]               # 矩阵取元素 [行, 列]
m[, 2]                # 取整列（drop 维度）
lst$name              # $ 取单个元素
lst[["scores"]]       # [[]] 取单个元素（去包装）
lst[1:2]              # [] 取子列表（保留 list 结构）
df[df$score > 86, ]   # 数据框按条件取行

# ============================================================
# 控制流：if / else / for / while / repeat
# ============================================================
if (x > 5) {
  print("big")
} else if (x == 5) {
  print("equal")
} else {
  print("small")
}

for (i in 1:3) print(paste("i =", i))

n <- 0
while (n < 3) {
  n <- n + 1
}

# repeat 必须手动 break
counter <- 0
repeat {
  counter <- counter + 1
  if (counter >= 2) break
}

# ============================================================
# 函数：function / 默认参数 / 椭圆 ...
# ============================================================
add <- function(a, b = 1) {       # 默认参数
  s <- a + b
  return(s)                       # 显式 return；否则返回最后一个表达式
}

greet <- function(name) paste("Hello,", name)

sum_all <- function(...) sum(...)  # ... 转发可变参数

# ============================================================
# apply 族：替代显式循环
# ============================================================
lapply(1:3, function(i) i^2)      # 返回 list
sapply(1:3, function(i) i^2)      # 自动简化为 vector
apply(m, 1, sum)                  # 按行（MARGIN = 1）应用
apply(m, 2, mean)                 # 按列（MARGIN = 2）应用

# ============================================================
# 常用统计：mean / median / sd / summary
# ============================================================
nums <- c(4, 7, 1, 9, 3, NA)
mean(nums)                        # 含 NA 返回 NA
mean(nums, na.rm = TRUE)          # 去除 NA 后求均值
median(nums, na.rm = TRUE)
sd(nums, na.rm = TRUE)
summary(nums)                     # 五数概括 + 均值
summary(df)                       # 数据框各列摘要

# ============================================================
# 绘图（plot 简要，避免运行时弹窗）
# ============================================================
# plot(1:5, (1:5)^2, type = "b", main = "y = x^2",
#      xlab = "x", ylab = "y", col = "blue")
# hist(rnorm(1000))               # 直方图
# boxplot(score ~ name, data = df)

# ============================================================
# 包：library / require
# ============================================================
# install.packages("dplyr")       # 安装（命令行运行一次即可）
# library(dplyr)                  # 加载到会话；找不到会报错
# require(dplyr)                  # 找不到返回 FALSE（不报错）
# if (requireNamespace("ggplot2", quietly = TRUE)) {
#   library(ggplot2)
# }

# ============================================================
# 管道：原生 |> （R 4.1+）与 magrittr %>%
# ============================================================
# 原生管道 |> ：左侧结果作为右侧函数的【第一个参数】
result_pipe <- 1:5 |> mean()      # 等价于 mean(1:5)

# _ 占位符（R 4.2+）：当左侧不应作为首参时，用 _ 显式指定位置
# mtcars |> lm(mpg ~ wt, data = _)   # 等价于 lm(mpg ~ wt, data = mtcars)

# magrittr %>% ：tidyverse 早期约定，行为类似但更宽松
# library(magrittr)
# mtcars %>% subset(cyl == 4) %>% nrow()
```
