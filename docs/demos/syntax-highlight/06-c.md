# C 语法高亮示例

以下 `c.c` 示例使用 ` ```c ` 围栏代码块渲染：

```c
/* C 语言：底层、贴近硬件的过程式语言，手动内存管理，广泛用于系统编程。 */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* ===== 宏定义 ===== */
#define MAX_LEN 100
#define SQUARE(x) ((x) * (x))

/* ===== 函数原型（声明在前） ===== */
int add(int a, int b);
void greet(const char *name);

/* ===== 结构体 + typedef ===== */
typedef struct {
    char name[32];
    int age;
} Person;

/* ===== 函数定义 ===== */
int add(int a, int b) {
    return a + b;
}

void greet(const char *name) {
    printf("Hello, %s!\n", name);
}

int main(void) {
    /* 变量与基本类型 */
    int i = 10;
    double d = 3.14;
    char c = 'A';
    float f = 2.71f;
    printf("i=%d d=%.2f c=%c f=%.2f\n", i, d, c, f);

    /* 运算符（含位运算） */
    int a = 12, b = 10;
    printf("a&b=%d a|b=%d a^b=%d a<<2=%d\n", a & b, a | b, a ^ b, a << 2);

    /* 控制流：if / else */
    if (i > 5) {
        printf("big\n");
    } else {
        printf("small\n");
    }

    /* switch */
    switch (c) {
        case 'A': printf("A grade\n"); break;
        case 'B': printf("B grade\n"); break;
        default:  printf("other\n");
    }

    /* for / while */
    for (int k = 0; k < 3; k++) {
        printf("k=%d\n", k);
    }
    int n = 2;
    while (n > 0) {
        n--;
    }

    /* 指针与解引用 */
    int x = 42;
    int *px = &x;
    *px = 99;
    printf("x = %d\n", x);

    /* 数组与 C 字符串 */
    int arr[5] = {1, 2, 3, 4, 5};
    int total = 0;
    for (int k = 0; k < 5; k++) {
        total += arr[k];
    }
    printf("arr sum = %d\n", total);

    char str[] = "C language";
    printf("len = %zu\n", strlen(str));

    /* 结构体 */
    Person p;
    strncpy(p.name, "Alice", sizeof(p.name) - 1);
    p.name[sizeof(p.name) - 1] = '\0';
    p.age = 30;
    printf("%s is %d\n", p.name, p.age);

    /* 动态内存：malloc / free */
    int *dyn = malloc(5 * sizeof(int));
    if (dyn == NULL) {
        return 1;
    }
    for (int k = 0; k < 5; k++) {
        dyn[k] = k * k;
    }
    free(dyn); // 必须手动释放

    /* 宏 */
    printf("SQUARE(5) = %d\n", SQUARE(5));

    /* 文件 I/O：fopen / fprintf / fgets / fclose */
    FILE *fp = fopen("demo.txt", "w");
    if (fp) {
        fprintf(fp, "line %d\n", 1);
        fclose(fp);
    }

    fp = fopen("demo.txt", "r");
    if (fp) {
        char buf[MAX_LEN];
        while (fgets(buf, sizeof(buf), fp) != NULL) {
            printf("read: %s", buf);
        }
        fclose(fp);
    }

    greet("World");
    printf("add(3,4) = %d\n", add(3, 4));
    return 0;
}
```
